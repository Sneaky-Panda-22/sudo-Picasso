  99 // Constructor for 2D array
 100 NumpyArray::NumpyArray(int r, int c) {
 101 
 102     if (r < 0 || c < 0) {
 103         throw std::invalid_argument("Array dimensions cannot be negative");
 104     }
 105 
 106     // Prevent overflow while calculating total size
 107     if (r != 0 && c > INT_MAX / r) {
 108         throw std::overflow_error("Array is too large");
 109     }
 110 
 111     rows = r;
 112     cols = c;
 113     totalSize = r * c;
 114     dimensions = 2;
 115 
 116     data = new double[totalSize];
 117 
 118     // Initialise the array with zeros
 119     for (int i = 0; i < totalSize; i++) {
 120         data[i] = 0.0;
 121     }
 122 }

 149 // Assignment operator
 150 NumpyArray& NumpyArray::operator=(const NumpyArray& obj) {
 151 
 152     // Prevent self-assignment (eg: a = a)
 153     if (this == &obj) {
 154         return *this;
 155     }
 156 
 157     // Allocate new memory and copy data inside
 158     double* newData = new double[obj.totalSize];
 159 
 160     for (int i = 0; i < obj.totalSize; i++) {
 161         newData[i] = obj.data[i];
 162     }
 163 
 164     // Clean old memory
 165     delete[] data;
 166 
 167     // Update dimensions and data
 168     rows = obj.rows;
 169     cols = obj.cols;
 170     totalSize = obj.totalSize;
 171     dimensions = obj.dimensions;
 172     data = newData;
 173 
 174     return *this;
 175 }

 271 double NumpyArray::min() const {
 272 
 273     if (totalSize == 0) {

 341 // Create an array containing a sequence of numbers
 342 NumpyArray NumpyArray::arange(int start, int stop, int step) {
 343 
 344     if (step <= 0) {
 345         throw std::invalid_argument("Step must be positive");
 346     }
 347 
 348     long long count = 0;
 349 
 350     if (start < stop) {
 351         count = ((long long)stop - start + step - 1) / step;
 352     }
 353     if (count > INT_MAX) {
 354         throw std::overflow_error("Array is too large");
 355     }
 356     NumpyArray result((int)count);
 357     long long value = start;
 358     for (int i = 0; i < result.size(); i++) {
 359         result(i) = value;
 360         value += step;
 361     }
 362     return result;
 363 }

 366 // Extract a portion of a 2D array
 367 NumpyArray NumpyArray::slice(int rowStart, int rowStop, int colStart, int colStop) const {
 368 
 369     // Check for invalid slice indices
 370     if (rowStart < 0 || rowStop > rows || rowStart >= rowStop ||
 371         colStart < 0 || colStop > cols || colStart >= colStop) {
 372 
 373         throw std::invalid_argument("Invalid slice indices");
 374     }
 375 
 376     int newRows = rowStop - rowStart;
 377     int newCols = colStop - colStart;
 378 
 379     NumpyArray result(newRows, newCols);
 380 
 381     for (int i = rowStart; i < rowStop; i++) {
 382         for (int j = colStart; j < colStop; j++) {
 383 
 384             result(i - rowStart, j - colStart) = data[i * cols + j];
 385         }
 386     }
 387 
 388     return result;
 389 }

 499 // Set RGB values of a pixel

 503         throw std::out_of_range("Pixel coordinates out of bounds");

 507     if (r < 0) r = 0;

 514     if (b > 255) b = 255;

 545 // Class to handle image input/output operations
 546 class ImageIO{
 547 public:
 548     //We don't need to create an ImageIO object just to load an image. Thus we make it static.
 549     // it returns an image.
 550     static Image loadPPM(std::string);
 551     static void display(const Image&);
 552 };
 553 
 554 // Load a PPM image from a file
 555 Image ImageIO::loadPPM(std::string filename) {
 556 
 557     std::ifstream file(filename);
 558 
 559     // Check whether file opened successfully
 560     if (!file) {
 561         throw std::runtime_error("Could not open image file");
 562     }
 563 
 564     std::string format;
 565     int width, height, maxValue;
 566 
 567     // Read PPM header
 568     file >> format;
 569 
 570     if (format != "P3") {
 571         throw std::runtime_error("Only P3 PPM format is supported");
 572     }
 573 
 574     file >> width >> height;
 575     file >> maxValue;
 576 
 577     if (!file || width <= 0 || height <= 0 ||
 578         maxValue <= 0 || maxValue > 65535) {
 579         throw std::runtime_error("Invalid PPM header");
 580     }
 581 
 582     // Create image object
 583     Image img(width, height);
 584 
 585     // Read pixel data
 586     for (int i = 0; i < height; i++) {
 587 
 588         for (int j = 0; j < width; j++) {
 589 
 590             int r, g, b;
 591 
 592             file >> r >> g >> b;
 593 
 594             if (!file || r < 0 || r > maxValue ||
 595                 g < 0 || g > maxValue ||
 596                 b < 0 || b > maxValue) {
 597 
 598                 throw std::runtime_error("Invalid pixel data");
 599             }
 600 
 601             // Convert values to the 0-255 range
 602             r = (r * 255 + maxValue / 2) / maxValue;
 603             g = (g * 255 + maxValue / 2) / maxValue;
 604             b = (b * 255 + maxValue / 2) / maxValue;
 605 
 606             img.setPixel(i, j, r, g, b);
 607         }
 608     }
 609 
 610     return img;
 611 }

 613 // Display the image directly in the terminal using ANSI escape codes
 614 void ImageIO::display(const Image& img) {
 615 
 616     int width = img.getWidth();
 617     int height = img.getHeight();
 618 
 619     // Maximum terminal display dimensions
 620     int maxWidth = 100;
 621     int maxHeight = 35;
 622 
 623     // Calculate scaling factor
 624     double scale = 1.0;
 625 
 626     if (width > maxWidth) {
 627         scale = (double)maxWidth / width;
 628     }
 629 
 630     if (height * scale > maxHeight) {
 631         scale = (double)maxHeight / height;
 632     }
 633 
 634     // Calculate new dimensions
 635     int newWidth = (int)(width * scale);
 636     int newHeight = (int)(height * scale);
 637 
 638     if (newWidth < 1) newWidth = 1;
 639     if (newHeight < 1) newHeight = 1;
 640 
 641     // Display the scaled image
 642     for (int i = 0; i < newHeight; i++) {
 643 
 644         for (int j = 0; j < newWidth; j++) {
 645 
 646             // Find corresponding pixel in original image
 647             int originalRow = i * height / newHeight;
 648             int originalCol = j * width / newWidth;
 649 
 650             std::vector<double> pixel =
 651                 img.getPixel(originalRow, originalCol);
 652 
 653             int r = pixel[0];
 654             int g = pixel[1];
 655             int b = pixel[2];
 656 
 657             std::cout << "\033[48;2;"
 658                       << r << ";"
 659                       << g << ";"
 660                       << b << "m  \033[0m";
 661         }
 662 
 663         std::cout << std::endl;
 664     }
 665 }

 667 // filters:
 668 // greyscale,brightness,invert. these perform different opeartions
 669 // but share a common idea ie:
 670 // they take an image, modify it and produce a processed image
 671 // thus instead of implementing every filter, we make a common class 
 672 // named ImgeFilter and add a pure virtual function to make it abstract!
 673 // can also show polymorphism!
 674 class ImageFilter{
 675 public:
 676     virtual void apply(Image&) = 0; // pure virtual function
 677     virtual ~ImageFilter() {} // virtual destructor
 678 };
 679 
 680 /* to convert into grayscale, the intensity value ultimately 
 681    depends on the three channels as:
 682 
 683         Gray = 0.299*Red + 0.587*Green + 0.114*Blue
 684 
 685    this is mainly because naturally the eyes are more sensitive to green
 686    as compared to red and more sensitive to red as compared to blue.
 687    so the green contributes more to the perceived brightness!*/
 688 
 689 class GrayScaleFilter : public ImageFilter{
 690 public:
 691     void apply(Image&)override;
 692 };
 693 
 694 // Apply the grayscale filter to the image
 695 void GrayScaleFilter::apply(Image& img) {
 696 
 697     int width = img.getWidth();
 698     int height = img.getHeight();
 699 
 700     for (int i = 0; i < height; i++) {
 701 
 702         for (int j = 0; j < width; j++) {
 703 
 704             std::vector<double> pixel = img.getPixel(i, j);
 705 
 706             double r = pixel[0];
 707             double g = pixel[1];
 708             double b = pixel[2];
 709 
 710             // Calculate the grayscale intensity using the standard luminosity formula
 711             double gray = 0.299 * r + 0.587 * g + 0.114 * b;
 712 
 713             img.setPixel(i, j, gray, gray, gray);
 714         }
 715     }
 716 }

 719 /* implementing Brightness Filter:
 720    to implement brightness filter, we just have to add a constant
 721    to our existing channels as: let 'k' be the constant then:
 722 
 723    c'(new channel) = c(original channel) + k  for all channels
 724 
 725 */
 726 
 727 class BrightnessFilter : public ImageFilter {
 728 private:
 729     double brightness;
 730 public:
 731     BrightnessFilter(double b) : brightness(b) {}
 732     void apply(Image&) override;
 733 };

 735 // Apply the brightness filter to the image
 736 void BrightnessFilter::apply(Image& img) {
 737 
 738     int width = img.getWidth();
 739     int height = img.getHeight();
 740 
 741     for (int i = 0; i < height; i++) {
 742 
 743         for (int j = 0; j < width; j++) {
 744 
 745             std::vector<double> pixel = img.getPixel(i, j);
 746 
 747             double r = pixel[0] + brightness;
 748             double g = pixel[1] + brightness;
 749             double b = pixel[2] + brightness;
 750 
 751             // Clamp RGB values between 0 and 255
 752             if (r > 255) r = 255;
 753             if (r < 0) r = 0;
 754 
 755             if (g > 255) g = 255;
 756             if (g < 0) g = 0;
 757 
 758             if (b > 255) b = 255;
 759             if (b < 0) b = 0;
 760 
 761             img.setPixel(i, j, r, g, b);
 762         }
 763     }
 764 }

 766 // invert filter:
 767 // intuitively its just the complement of channels
 768 // mathematically, c'(new channel) = 255 - c(old channel) 
 769 class InvertFilter : public ImageFilter {
 770 public:
 771     void apply(Image&) override;

 774 // Apply the invert filter to the image
 775 void InvertFilter::apply(Image& img) {
 776 
 777     int width = img.getWidth();
 778     int height = img.getHeight();
 779 
 780     for (int i = 0; i < height; i++) {
 781 
 782         for (int j = 0; j < width; j++) {
 783 
 784             std::vector<double> pixel = img.getPixel(i, j);
 785 
 786             // Invert the channel by subtracting its value from 255
 787             double r = 255 - pixel[0];
 788             double g = 255 - pixel[1];
 789             double b = 255 - pixel[2];
 790 
 791             img.setPixel(i, j, r, g, b);
 792         }
 793     }

 816 // Constructor for CropTransform
 817 CropTransform::CropTransform(int rs,int re,int cs,int ce) : rowStart(rs),rowStop(re),colStart(cs),colStop(ce) {}

 819 // Apply crop transformation to the image
 820 void CropTransform::apply(Image& img) {
 821     // validating coordinates
 822     if (rowStart < 0 || rowStop > img.getHeight() ||
 823         colStart < 0 || colStop > img.getWidth() ||
 824         rowStart >= rowStop || colStart >= colStop) {
 825 
 826         throw std::out_of_range("Invalid cropping dimensions");
 827     }
 828 
 829     // cropping individual RGB channels
 830     NumpyArray croppedRed = img.getRedChannel().slice(rowStart,rowStop,colStart,colStop);
 831     NumpyArray croppedGreen = img.getGreenChannel().slice(rowStart,rowStop,colStart,colStop);
 832     NumpyArray croppedBlue = img.getBlueChannel().slice(rowStart,rowStop,colStart,colStop);
 833 
 834     // creating cropped image
 835     Image cropped(colStop-colStart,rowStop-rowStart,croppedRed,croppedGreen,croppedBlue);
 836 
 837     // replace the origiinal image
 838     img = cropped;

 855 // Apply flip transformation to the image
 856 void FlipTransform::apply(Image& img) {
 857     int width = img.getWidth();
 858     int height = img.getHeight();
 859 
 860     Image flipped(width,height);
 861 
 862     for(int i=0;i<height;i++){
 863         for(int j=0;j<width;j++){
 864             std::vector<double> pixel = img.getPixel(i,j);
 865             // Remap coordinates based on the selected axis
 866             if(axis==0) flipped.setPixel(height-i-1,j,pixel[0],pixel[1],pixel[2]);
 867             else flipped.setPixel(i,width-j-1,pixel[0],pixel[1],pixel[2]);
 868         }
 869     }
 870     img = flipped;
 871 }

 886 }

 888 // Apply rotation transformation to the image
 889 void RotateTransform::apply(Image& img) {
 890 
 891     int width = img.getWidth();
 892     int height = img.getHeight();
 893 
 894     int newWidth = width;
 895     int newHeight = height;
 896 
 897     // Swap dimensions for 90 and 270 degree rotations
 898     if (angle == 90 || angle == 270) {
 899         newWidth = height;
 900         newHeight = width;
 901     }
 902 
 903     Image rotated(newWidth, newHeight);
 904 
 905     for (int i = 0; i < height; i++) {
 906         for (int j = 0; j < width; j++) {
 907 
 908             std::vector<double> pixel = img.getPixel(i, j);
 909 
 910             int newRow, newCol;
 911 
 912             // Determine new coordinates based on rotation angle
 913             if (angle == 90) {
 914                 newRow = j;
 915                 newCol = height - 1 - i;
 916             }
 917             else if (angle == 180) {
 918                 newRow = height - 1 - i;
 919                 newCol = width - 1 - j;
 920             }
 921             else {
 922                 newRow = width - 1 - j;
 923                 newCol = i;
 924             }
 925 
 926             rotated.setPixel(newRow, newCol,
 927                              pixel[0], pixel[1], pixel[2]);
 928         }
 929     }
 930 
 931     img = rotated;
 932 }

 964 // Calculate the overall average brightness of the image
 965 double ImageAnalyzer::getAverageBrightness(const Image& img) {
 966 
 967     double total = 0;
 968 
 969     for (int i = 0; i < img.getHeight(); i++) {
 970         for (int j = 0; j < img.getWidth(); j++) {
 971

 973 
 974             double brightness =
 975                 0.299 * pixel[0] +
 976                 0.587 * pixel[1] +
 977                 0.114 * pixel[2];
 978 
 979             total += brightness;
 980         }
 981     }
 982 
 983     int totalPixels = img.getWidth() * img.getHeight();
 984 
 985     return total / totalPixels;
